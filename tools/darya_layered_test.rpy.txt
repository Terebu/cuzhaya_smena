    def cs_qa_darya_heads(start):
        tiles=[]
        entries=cs_dasha_metadata["pioneer"][20:]
        for n,e in enumerate(entries[start:start+6]):
            actor=renpy.get_registered_image("cs_darya "+e["pose"]+" pioneer normal")
            head=Transform(actor,crop=(280,0,440,420),size=(300,286))
            tiles.extend([(n%3*420,n//3*350),head])
            tiles.extend([(n%3*420+10,n//3*350+290),Text(e["pose"],size=24,color="#ffffff")])
        return LiveComposite((1280,720),*tiles)
