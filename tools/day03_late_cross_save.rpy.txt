init 99 python:
    renpy.cs_lx_fast=False
    cs_lx_old_say=config.say_arguments_callback
    cs_lx_old_menu=config.menu_arguments_callback
    cs_lx_old_labels=config.label_callback
    def cs_lx_say(who,*args,**kwargs):
        if cs_lx_old_say:args,kwargs=cs_lx_old_say(who,*args,**kwargs)
        if renpy.cs_lx_fast:kwargs["interact"]=False
        return args,kwargs
    def cs_lx_menu(items,*args,**kwargs):
        label,val=next((l,v) for l,v in items if v is not None)
        return val() if callable(val) else val
    def cs_lx_menu_args(*args,**kwargs):
        if cs_lx_old_menu:args,kwargs=cs_lx_old_menu(*args,**kwargs)
        if renpy.cs_lx_fast:store.menu=cs_lx_menu
        return args,kwargs
    def cs_lx_labels(name,abnormal):
        if cs_lx_old_labels:cs_lx_old_labels(name,abnormal)
        if renpy.cs_lx_fast:
            with open(config.basedir+"/late_cross_progress.txt","a") as trace:
                trace.write(name+"\n")
            assert name!="cs_day03_121","Old lunch must not replay"
    config.say_arguments_callback=cs_lx_say
    config.menu_arguments_callback=cs_lx_menu_args
    config.label_callback=cs_lx_labels
