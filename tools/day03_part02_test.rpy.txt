# -*- coding: utf-8 -*-
init 95 python:
    def cs_day03_p2_capture(name):
        renpy.screenshot(config.basedir + "/p2_%s_%s.png" % (renpy.cs_p2_case, name))

label cs_qa_day03_p2:
    $ cs_reset_state()
    $ cs_active = True
    $ cs_darya_known = True
    $ cs_slavya_known = True
    $ cs_lena_elbow_bandaged = True
    $ cs_lena_elbow_injury_day2 = 2
    $ cs_lena_no_sport_day2 = True
    $ cs_friend_sl = renpy.cs_p2_poster
    $ set_mode_adv()
    $ day_time()
    $ persistent.sprite_time = "day"
    if renpy.cs_p2_full:
        $ cs_friend_se = renpy.cs_p2_initial
        jump cs_day03
    $ cs_friend_se = renpy.cs_p2_initial
    $ cs_sl_poster_saved_day3 = bool(renpy.cs_p2_poster)
    $ cs_day03_poster_location = "slavya" if cs_sl_poster_saved_day3 else None
    if renpy.cs_p2_reach:
        call cs_day02_semen_choice
        call cs_day02_road_choice
    if not renpy.cs_p2_reach:
        jump cs_qa_day03_p2_choice_route
    jump cs_day03_part02

label cs_qa_day03_p2_choice_route:
    $ cs_day03_part01_done = True
    call cs_day03_86
    call cs_day03_87
    call cs_day03_88
    $ cs_day03_part02_done = True
    jump cs_day03_part03_pending

label cs_qa_day03_p2_replay:
    hide screen cs_day03_part02_wait_screen
    show cs_semen neutral pioneer normal at cs_semen_normal_left
    $ cs_frame_focus(["cs_darya", "cs_semen"])
    call cs_day03_semen_choice
    jump cs_day03_part03_pending

testcase cs_day03_p2_0:
    pause 0.5
    $ renpy.cs_p2_case = 0
    $ renpy.cs_p2_initial = 0
    $ renpy.cs_p2_ask = False
    $ renpy.cs_p2_poster = 0
    $ renpy.cs_p2_full = False
    $ renpy.cs_p2_reach = False
    $ persistent.font_size = "small"
    $ renpy.set_physical_size((1280,720))
    $ _preferences.text_cps = 0
    $ _test.force = True
    $ _test.timeout = 60.0
    run Start("cs_qa_day03_p2")
    click until label cs_check_day03_86_semen
    pause 0.5
    assert renpy.showing("cs_darya") and renpy.showing("cs_semen")
    assert renpy.get_image_bounds("cs_semen")[0] < renpy.get_image_bounds("cs_darya")[0]
    $ renpy.cs_p2_pair_bounds = [renpy.get_image_bounds(t) for t in ("cs_darya","cs_semen")]
    $ cs_day03_p2_capture("pair")
    $ renpy.take_screenshot()
    $ renpy.save("p2_before_choice_"+str(renpy.cs_p2_case))
    $ renpy.load("p2_before_choice_"+str(renpy.cs_p2_case))
    pause 0.4
    click until label cs_check_day03_87_menu
    pause 0.5
    assert renpy.showing("cs_darya") and renpy.showing("cs_semen")
    $ cs_day03_p2_capture("choice")
    "Не задерживать его перед завтраком"
    click until label cs_check_day03_87b
    pause 0.5
    assert renpy.showing("cs_darya") and renpy.showing("cs_semen")
    assert cs_friend_se == renpy.cs_p2_initial + (2 if renpy.cs_p2_reach else 0) + int(renpy.cs_p2_ask)
    assert cs_se_day3_morning_awarded == bool(renpy.cs_p2_ask)
    $ cs_day03_p2_capture("branch")
    $ renpy.take_screenshot()
    $ renpy.save("p2_after_choice_"+str(renpy.cs_p2_case))
    $ renpy.load("p2_after_choice_"+str(renpy.cs_p2_case))
    pause 0.4
    assert cs_friend_se == renpy.cs_p2_initial + (2 if renpy.cs_p2_reach else 0) + int(renpy.cs_p2_ask)
    click until label cs_check_day03_88_semen_left
    pause 0.5
    assert renpy.showing("cs_darya") and not renpy.showing("cs_semen")
    $ cs_day03_p2_capture("semen_left")
    click until label cs_check_day03_88_compliment
    pause 0.5
    assert renpy.showing("cs_darya")
    $ cs_day03_p2_capture("compliment")
    click until label cs_check_day03_88_dining
    pause 0.5
    assert renpy.showing("cs_darya") and not renpy.showing("cs_semen")
    $ cs_day03_p2_capture("dining")
    click until label cs_check_day03_part02_complete
    assert cs_day03_part01_done and cs_day03_part02_done
    assert cs_lena_elbow_bandaged and cs_lena_elbow_injury_day2 == 2 and cs_lena_no_sport_day2
    assert cs_friend_sl == renpy.cs_p2_poster and cs_friend_mz == 0 and cs_friend_viola == 0
    $ cs_day03_p2_capture("end")
    $ open(config.basedir + "/p2_%s_PASS.txt" % renpy.cs_p2_case,"w").write("PASS")
    $ cs_reset_state()
    assert not cs_day03_part02_done and not cs_se_day3_morning_awarded
    run Quit(confirm=False)
