    def cs_p3_actor_opaque(tag):
        actor = renpy.game.context().scene_lists.get_displayable_by_tag("master", tag)
        return actor is not None and getattr(getattr(actor,"state",None),"alpha",1.0) >= 0.999
    def cs_p3_capture(name):
        renpy.screenshot(config.basedir + "/p3_%s_%s.png" % (renpy.cs_p3_case,name))
