testcase cs_p7_load_old_save:
    pause 0.5
    $ _preferences.text_cps = 0
    $ _preferences.transitions = 2
    $ _test.force = True
    $ _test.timeout = 180.0
    $ renpy.load("p7_cross_version")
    pause 0.5
    assert not bool(renpy.get_screen("cs_day03_part06_wait_screen"))
    assert cs_friend_un == 2 and cs_friend_mz == 3
    click until label cs_check_day03_125_menu
    "Предложить Лене Тихоновой пообедать вместе"

    pause 0.5
    assert cs_friend_un == 3 and cs_friend_mz == 3
    assert cs_un_sketch_plan_discussed_day3 and cs_un_day3_lunch_together

    assert cs_day03_documents_location == "administration" and cs_mz_book_requested_day3
    assert cs_lena_elbow_injury_day2 == 2 and cs_lena_no_sport_day2 and cs_viola_spare_bandage
    $ renpy.screenshot(config.basedir + "/p7_cross_version_loaded.png")
    $ open(config.basedir + "/p7_cross_save_PASS.txt","w").write("PASS")
    run Quit(confirm=False)
