    renpy.cs_p12_old_fast=False
    renpy.cs_p12_old_audio=False
    cs_x12_say_old=config.say_arguments_callback
    cs_x12_labels_old=config.label_callback
    def cs_x12_say(who,*args,**kwargs):
        if cs_x12_say_old:args,kwargs=cs_x12_say_old(who,*args,**kwargs)
        if renpy.cs_p12_old_fast:
            if renpy.cs_p12_old_audio and renpy.showing("bg ext_dining_hall_away_day"):
                renpy.cs_p12_old_fast=False
            else:kwargs["interact"]=False
        return args,kwargs
    def cs_x12_labels(name,abnormal):
        if cs_x12_labels_old:cs_x12_labels_old(name,abnormal)
        if renpy.cs_p12_old_fast and name=="cs_day03_161a":renpy.cs_p12_old_fast=False
    config.say_arguments_callback=cs_x12_say
    config.label_callback=cs_x12_labels
