init 120 python:
    renpy.cs_d5_visual=False
    def cs_d5_visual_capture(event,**kwargs):
        if not renpy.cs_d5_visual or event!='show_done':return
        renpy.pause(0.01,hard=True)
        sc=renpy.get_screen('cs_d5_insert')
        content=getattr(sc,'scope',{}).get('content','') if sc else ''
        key=(content,cs_d5_clock,bool(renpy.get_screen('cs_d5_notes')),tuple(renpy.get_attributes('bg') or ()))
        if key!=getattr(renpy,'cs_d5_visual_key',None):
            renpy.cs_d5_visual_key=key
            renpy.cs_d5_visual_frame+=1
            renpy.pause(0.3,hard=True)
            renpy.screenshot(config.basedir+'/day05_insert_%02d.png'%renpy.cs_d5_visual_frame)
    config.all_character_callbacks.append(cs_d5_visual_capture)
