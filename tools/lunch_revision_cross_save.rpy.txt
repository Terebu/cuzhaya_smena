init 96 python:
    renpy.cs_lc_fast=False
    renpy.cs_lc_at_menu=False
    cs_lc_old_args=config.say_arguments_callback
    cs_lc_old_label=config.label_callback
    def cs_lc_args(who,*args,**kwargs):
        if cs_lc_old_args: args,kwargs=cs_lc_old_args(who,*args,**kwargs)
        if renpy.cs_lc_fast: kwargs["interact"]=False
        return args,kwargs
    def cs_lc_label(name,abnormal):
        if cs_lc_old_label: cs_lc_old_label(name,abnormal)
        if name=="cs_check_day03_125_menu":
            renpy.cs_lc_fast=False
            renpy.cs_lc_at_menu=True
    config.say_arguments_callback=cs_lc_args
    config.label_callback=cs_lc_label
