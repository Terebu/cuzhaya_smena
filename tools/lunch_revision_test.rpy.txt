init 97 python:
    renpy.cs_lr_fast=False
    cs_lr_old_say=config.say_arguments_callback
    cs_lr_old_labels=config.label_callback
    cs_lr_old_menu=config.menu_arguments_callback
    def cs_lr_say(who,*args,**kwargs):
        if cs_lr_old_say:
            args,kwargs=cs_lr_old_say(who,*args,**kwargs)
        if renpy.cs_lr_fast: kwargs["interact"]=False
        return args,kwargs
    def cs_lr_menu(items,*args,**kwargs):
        choices=[(label,value) for label,value in items if value is not None]
        val=choices[0 if renpy.cs_lr_yes else -1][1]
        return val() if callable(val) else val
    def cs_lr_menu_args(*args,**kwargs):
        if cs_lr_old_menu: args,kwargs=cs_lr_old_menu(*args,**kwargs)
        if renpy.cs_lr_fast: store.menu=cs_lr_menu
        return args,kwargs
    def cs_lr_label(name,abnormal):
        if cs_lr_old_labels: cs_lr_old_labels(name,abnormal)
        if not renpy.cs_lr_fast: return
        if name in ("cs_day03_122","cs_day03_124","cs_day03_125","cs_day03_126","cs_day03_127","cs_check_day03_127_darya_left","cs_check_day03_128_library"):
            renpy.cs_lr_pending=name
        if name=="cs_day03_122": assert renpy.showing("cs_zhenya") and not renpy.showing("cs_darya")
        if name=="cs_day03_127": assert not renpy.showing("cs_zhenya") and cs_director_states["cs_zhenya"]=="ABSENT"
        if name=="cs_check_day03_127_darya_left": assert not renpy.showing("cs_darya") and cs_director_states["cs_darya"]=="ABSENT"
    def cs_lr_character(event,**kwargs):
        if event!="show_done" or not renpy.cs_lr_fast: return
        cast=tuple(sorted(t for t in cs_presence_tags if renpy.showing(t)))
        if cast!=renpy.cs_lr_cast or renpy.cs_lr_pending:
            renpy.cs_lr_cast=cast
            renpy.cs_lr_count+=1
            renpy.pause(0.25,hard=True)
            cs_p6_capture("%02d_%s"%(renpy.cs_lr_count,renpy.cs_lr_pending or "cast"))
            renpy.cs_lr_pending=None
    config.say_arguments_callback=cs_lr_say
    config.label_callback=cs_lr_label
    config.menu_arguments_callback=cs_lr_menu_args
    config.all_character_callbacks.append(cs_lr_character)
