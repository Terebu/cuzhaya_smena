init 94 python:
    renpy.cs_rev_cross_fast=False
    renpy.cs_rev_cross_observe=False
    cs_rev_cross_say_original=config.say_arguments_callback
    def cs_rev_cross_say(who,*args,**kwargs):
        if cs_rev_cross_say_original:args,kwargs=cs_rev_cross_say_original(who,*args,**kwargs)
        if renpy.cs_rev_cross_fast:kwargs["interact"]=False
        return args,kwargs
    config.say_arguments_callback=cs_rev_cross_say
    cs_rev_cross_labels_original=config.label_callback
    def cs_rev_cross_labels(name,abnormal):
        if cs_rev_cross_labels_original:cs_rev_cross_labels_original(name,abnormal)
        if renpy.cs_rev_cross_observe:
            with open(config.basedir+"/revision_cross_progress.txt","a") as f:f.write(name+"\n")
            if name=="cs_check_day03_104":renpy.cs_rev_cross_fast=False
    config.label_callback=cs_rev_cross_labels

testcase cs_revision_load_old_menu:
    pause 0.5
    $ persistent.font_size="small"
    $ renpy.set_physical_size((1280,720))
    $ _preferences.text_cps=0
    $ _preferences.transitions=2
    $ _test.force=True
    $ _test.timeout=120.0
    $ renpy.cs_rev_cross_observe=True
    $ renpy.cs_rev_cross_fast=True
    $ renpy.load("revision_previous_menu")
    pause 0.5
    $ renpy.screenshot(config.basedir+"/revision_cross_loaded_menu.png")
    $ open(config.basedir+"/revision_new_stack.txt","w").write(repr(renpy.game.context().return_stack))
    assert cs_friend_viola==1 and not cs_dv_medpoint_day3
