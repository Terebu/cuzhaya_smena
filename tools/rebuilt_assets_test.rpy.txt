    def cs_patch_say(who,*args,**kwargs):
        if cs_patch_say_original:
            args,kwargs=cs_patch_say_original(who,*args,**kwargs)
        if renpy.cs_patch_active:
            kwargs["interact"]=False
        return args,kwargs
    def cs_patch_menu(items,*args,**kwargs):
        values=[v for l,v in items if v is not None]
        value=values[0]
        return value() if callable(value) else value
    def cs_patch_menu_args(*args,**kwargs):
        if cs_patch_menu_original:
            args,kwargs=cs_patch_menu_original(*args,**kwargs)
        if renpy.cs_patch_active:
            store.menu=cs_patch_menu
        return args,kwargs
    def cs_patch_labels(name,abnormal):
        if cs_patch_label_original:
            cs_patch_label_original(name,abnormal)

                renpy.pause(0.2,hard=True)

            assert renpy.music.get_playing(channel="ambience")==target
        names=[" ".join((t,)+tuple(renpy.get_attributes(t) or ())) for t in renpy.get_showing_tags()]

    def cs_rebuilt_grid(costume,start):
        entries=[e for e in cs_dasha_metadata[costume] if e["pose"]==e["path"].split("/")[-1].replace("darya_"+costume+"_","").replace(".png","")]
        tiles=[]
        for n,e in enumerate(entries[start:start+6]):
            actor=renpy.get_registered_image("cs_darya "+e["pose"]+" "+costume+" normal")
            tiles.extend([(n%3*600,n//3*510),Transform(actor,crop=(220,0,540,450),size=(520,433))])
            tiles.extend([(n%3*600,n//3*510+440),Text(e["pose"],size=26,color="#fff")])
        return LiveComposite((1920,1080),*tiles)
