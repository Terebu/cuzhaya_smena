init 90 python:
    def cs_validate_asset_portability():
        import json
        manifest = json.load(renpy.file(cs_resolve_asset("qa_asset_manifest.json")))
        for rel in manifest:
            path = cs_resolve_asset(rel)
            if not renpy.loadable(path):
                raise Exception("Unreachable asset: " + path)
            with renpy.file(path) as f:
                if not f.read(1): raise Exception("Empty asset: " + path)
        with renpy.file(cs_resolve_asset("sp/dasha_metadata.json")) as f:
            metadata = json.load(f)
        for entries in metadata.values():
            for entry in entries:
                assert renpy.loadable(cs_resolve_asset(entry["path"]))
        assert renpy.has_image("cs_darya shrug pioneer normal")
        assert renpy.has_image("cs_semen neutral pioneer normal")
        assert renpy.has_image("cs_zhenya normal pioneer normal")
        assert renpy.has_image("cs custom serious medical normal")
        with open(config.basedir + "/asset_validation.json", "w") as f:
            json.dump(dict(count=len(manifest),root=cs_asset_prefix,searchpath=config.searchpath),f)
